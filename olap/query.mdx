-- KUERI 1 — PERTANYAAN BISNIS 1 (PB-01)
-- Pertanyaan: Bagaimana tren nilai penjualan online setiap bulan/tahun?
-- Tujuan Analisis: Mengamati pertumbuhan omzet, volume unit, dan transaksi per periode kalender.
-- =============================================================================

-- [MDX PB-01]
SELECT
  {[Measures].[Sales Amount], [Measures].[Order Quantity], [Measures].[Distinct Orders], [Measures].[Average Order Value]} ON COLUMNS,
  NON EMPTY [Waktu].[Tahun].Members ON ROWS
FROM [OnlineSalesCube];

-- [MDX PB-01 Detail Bulanan]
SELECT
  {[Measures].[Sales Amount], [Measures].[Order Quantity], [Measures].[Distinct Orders]} ON COLUMNS,
  NON EMPTY [Waktu].[Bulan].Members ON ROWS
FROM [OnlineSalesCube];

-- [SQL Pembanding PB-01]
SELECT 
  d.tahun,
  COUNT(DISTINCT f.sales_order_id) AS distinct_orders,
  SUM(f.order_qty) AS order_quantity,
  ROUND(SUM(f.line_total), 2) AS sales_amount,
  ROUND(SUM(f.line_total) / COUNT(DISTINCT f.sales_order_id), 2) AS average_order_value
FROM fact_online_sales f
JOIN dim_date d ON f.tanggal_key = d.tanggal_key
GROUP BY d.tahun
ORDER BY d.tahun;

-- KUERI 2 — PERTANYAAN BISNIS 2 (PB-02)
-- Pertanyaan: Kategori produk mana yang penjualannya paling besar?
-- Tujuan Analisis: Mengidentifikasi pilar produk penghasil omzet terbesar dan harga rata-rata unit.
-- =============================================================================

-- [MDX PB-02]
SELECT
  {[Measures].[Sales Amount], [Measures].[Order Quantity], [Measures].[Average Unit Price]} ON COLUMNS,
  ORDER([Produk].[Kategori].Members, [Measures].[Sales Amount], BDESC) ON ROWS
FROM [OnlineSalesCube];

-- [SQL Pembanding PB-02]
SELECT 
  p.category_name AS kategori,
  ROUND(SUM(f.line_total), 2) AS sales_amount,
  SUM(f.order_qty) AS order_quantity,
  ROUND(SUM(f.line_total) / SUM(f.order_qty), 2) AS average_unit_price
FROM fact_online_sales f
JOIN dim_product p ON f.product_key = p.product_key
GROUP BY p.category_name
ORDER BY sales_amount DESC;

-- Pertanyaan: Pelanggan mana yang memberikan kontribusi terbesar pada penjualan online?
-- Tujuan Analisis: Menemukan 10 pelanggan VIP teratas beserta frekuensi pesanan dan AOV.
-- =============================================================================

-- [MDX PB-03]
SELECT
  {[Measures].[Sales Amount], [Measures].[Distinct Orders], [Measures].[Average Order Value]} ON COLUMNS,
  TOPCOUNT([Pelanggan].[Pelanggan].Members, 10, [Measures].[Sales Amount]) ON ROWS
FROM [OnlineSalesCube];

-- [SQL Pembanding PB-03]
SELECT 
  c.customer_id,
  c.customer_name,
  ROUND(SUM(f.line_total), 2) AS sales_amount,
  COUNT(DISTINCT f.sales_order_id) AS distinct_orders,
  ROUND(SUM(f.line_total) / COUNT(DISTINCT f.sales_order_id), 2) AS average_order_value
FROM fact_online_sales f
JOIN dim_customer c ON f.customer_key = c.customer_key
GROUP BY c.customer_id, c.customer_name
ORDER BY sales_amount DESC
LIMIT 10;

-- KUERI 4 — PERTANYAAN BISNIS 4 (PB-04)
-- Pertanyaan: Wilayah mana yang mencatat transaksi dan nilai penjualan online terbesar?
-- Tujuan Analisis: Pemetaan omzet global berdasarkan 3 grup wilayah (North America, Pacific, Europe).
-- =============================================================================

-- [MDX PB-04]
SELECT
  {[Measures].[Sales Amount], [Measures].[Distinct Orders], [Measures].[Average Order Value]} ON COLUMNS,
  ORDER([Wilayah].[Grup Wilayah].Members, [Measures].[Sales Amount], BDESC) ON ROWS
FROM [OnlineSalesCube];

-- [SQL Pembanding PB-04]
SELECT 
  t.territory_group AS grup_wilayah,
  ROUND(SUM(f.line_total), 2) AS sales_amount,
  COUNT(DISTINCT f.sales_order_id) AS distinct_orders,
  ROUND(SUM(f.line_total) / COUNT(DISTINCT f.sales_order_id), 2) AS average_order_value
FROM fact_online_sales f
JOIN dim_territory t ON f.territory_key = t.territory_key
GROUP BY t.territory_group
ORDER BY sales_amount DESC;

-- KUERI 5 — PERTANYAAN BISNIS 5 (PB-05)
-- Pertanyaan: Bagaimana distribusi nilai penjualan online menurut penawaran/promosi?
-- Tujuan Analisis: Mengukur efektivitas program diskon dan perbandingan penjualan normal vs promo.
-- =============================================================================

-- [MDX PB-05]
SELECT
  {[Measures].[Sales Amount], [Measures].[Order Quantity], [Measures].[Total Rows]} ON COLUMNS,
  ORDER([Promosi].[Penawaran].Members, [Measures].[Sales Amount], BDESC) ON ROWS
FROM [OnlineSalesCube];

-- [SQL Pembanding PB-05]
SELECT 
  p.special_offer_id,
  p.description AS program_promosi,
  p.discount_pct,
  ROUND(SUM(f.line_total), 2) AS sales_amount,
  SUM(f.order_qty) AS order_quantity,
  COUNT(*) AS total_rows
FROM fact_online_sales f
JOIN dim_promotion p ON f.promotion_key = p.promotion_key
GROUP BY p.special_offer_id, p.description, p.discount_pct
ORDER BY sales_amount DESC;

-- KUERI 6 — PERTANYAAN BISNIS 6 (PB-06)
-- Pertanyaan: Bagaimana kelengkapan informasi kurs & distribusi valuta transaksi?
-- Tujuan Analisis: Memvalidasi proporsi pesanan dengan kurs vs tanpa kurs (default USD).
-- =============================================================================

-- [MDX PB-06]
SELECT
  {[Measures].[Sales Amount], [Measures].[Distinct Orders]} ON COLUMNS,
  NON EMPTY [Mata Uang].[Mata Uang Tujuan].Members ON ROWS
FROM [OnlineSalesCube];

-- [SQL Pembanding PB-06]
SELECT 
  CASE WHEN c.currency_rate_id = -1 THEN 'Tanpa Kurs (Default USD)' ELSE 'Dengan Kurs' END AS status_kurs,
  c.to_currency_code AS mata_uang_tujuan,
  COUNT(DISTINCT f.sales_order_id) AS distinct_orders,
  ROUND(SUM(f.line_total), 2) AS sales_amount
FROM fact_online_sales f
JOIN dim_currency c ON f.currency_key = c.currency_key
GROUP BY status_kurs, c.to_currency_code
ORDER BY distinct_orders DESC;

-- KUERI 7 — PERTANYAAN BISNIS 7 (PB-07)
-- Pertanyaan: Bagaimana sebaran pesanan online menurut metode pengiriman?
-- Tujuan Analisis: Mengaudit metode logistik yang digunakan pada seluruh pesanan online.
-- =============================================================================

-- [MDX PB-07]
SELECT
  {[Measures].[Sales Amount], [Measures].[Distinct Orders], [Measures].[Order Quantity]} ON COLUMNS,
  [Metode Kirim].[Metode Pengiriman].Members ON ROWS
FROM [OnlineSalesCube];

-- [SQL Pembanding PB-07]
SELECT 
  s.ship_method_id,
  s.ship_method_name,
  ROUND(SUM(f.line_total), 2) AS sales_amount,
  COUNT(DISTINCT f.sales_order_id) AS distinct_orders,
  SUM(f.order_qty) AS order_quantity
FROM fact_online_sales f
JOIN dim_ship_method s ON f.ship_method_key = s.ship_method_key
GROUP BY s.ship_method_id, s.ship_method_name
ORDER BY sales_amount DESC;
